#INVOKE PRISMA CLI
npx prisma

#CREATE PRISMA SCHEMAFILE
npx prisma init

#NEW MIGRATION (CHANGES ON DATABASE SCHEMA)
npx prisma migrate dev --name user-table-changed

#APPLY LATEST CHANGES ON DATABASE SCHEMA (APPLIES ALL THE MIGRATIONS DONE BY ANYONE)
npx prisma migrate dev

#APPLY ESPECIFIC MIGRATION
npx prisma migrate dev --name 'third_migration_name'

#RESET MIGRATION
npx prisma migrate reset


#UPLOAD THE DATAMODEL IN PRODUCTION
npx prisma migrate up --experimental
